\section{Tabelle da modificare}

\subsection{Agenti}
Tabella degli agenti esterni dell'azienda, sia che utilizzino o meno il programma.
Può essere accorpata alla tabella degli utenti con opportune accortezze.
\begin{lstlisting}

    CREATE TABLE [dbo].[agenti] (
    [age_id]         INT            IDENTITY (1, 1) NOT NULL,
    [age_nome]       NVARCHAR (40)  NULL,
    [age_indirizzo]  NVARCHAR (150) NULL,
    [age_cap]        NVARCHAR (5)   NULL,
    [age_citta]      NVARCHAR (25)  NULL,
    [age_provincia]  NVARCHAR (2)   NULL,
    [age_zona]       NVARCHAR (20)  NULL,
    [age_telefono]   NVARCHAR (15)  NULL,
    CONSTRAINT [Agente$PrimaryKey] PRIMARY KEY CLUSTERED ([age_id] ASC)
); 
\end{lstlisting}
